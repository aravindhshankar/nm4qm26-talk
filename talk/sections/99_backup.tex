\begin{frame}
\ft{Backup: universal two-spin entropy curves}
\small
\begin{itemize}
 \item Antiferromagnetic, $J>0$: $S(J{=}T)=\ln3+\ldots$ (universal function of $u=J/4T$)
 \item Ferromagnetic, $J<0$: spin-1 plateau, $S\to\ln 3$
 \item Kondo: $S(T)\simeq \ln 2/(1+T_K/T)$, with $S(T_K)=\tfrac12\ln2$
\end{itemize}
\end{frame}

\begin{frame}
\ft{Backup: perturbative RKKY, sign from the NLDOS}
\small
\[ \rho(\mathbf r,\omega)=\rho_s(\mathbf r,\omega)+\rho_a(\mathbf r,\omega),\quad \rho_{s/a}(-\omega)=\pm\rho_{s/a}(\omega) \]
\begin{itemize}
 \item $J_{\rm RKKY}\propto J^2$ from the one-loop spin susceptibility
 \item Positive denominator: sign set by the symmetric vs antisymmetric NLDOS
 \item Same sublattice: symmetric dominates, FM; opposite: AFM
\end{itemize}
\end{frame}
